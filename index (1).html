<script>
(() => {

let speedMultiplier = 1.0;

class Particle {

this.x = originX + (Math.random() - 0.5) * spread;
this.y = originY + (Math.random() - 0.5) * 8;
this.vx = (Math.random() - 0.5) * 0.4;

this.y -= this.life * 1.2;
}
}

this.phase += 0.06 * m;
const sway = Math.sin(this.phase) * this.wobble * m;
this.vx += sway;
this.vx *= 0.95;
this.x += this.vx * m;
this.y += this.vy * m;
this.life++;

this.currentSize = this.size * (1 - lifeRatio * 0.6);
this.currentWidth = this.currentSize * this.widthRatio;
}

let hue, sat, light, alpha;
if (lifeRatio < 0.2) {

} else if (lifeRatio < 0.5) {

sat = 100;
light = 60 - (lifeRatio - 0.2) * 40;
alpha = 0.75;
} else {

light = 35 - (lifeRatio - 0.5) * 30;
alpha = 0.5 * (1 - (lifeRatio - 0.5) / 0.5);
}
ctx.save();
ctx.translate(this.x, this.y);
const grad = ctx.createRadialGradient(0, 0, 0, 0, 0,
this.currentSize);
grad.addColorStop(0, `hsla(${hue},${sat}%,${light}%,${alpha})`);
grad.addColorStop(0.5, `hsla(${hue - 5},${sat}%,${light *
0.8}%,${alpha * 0.7})`);

0.5}%,0)`);

ctx.ellipse(0, 0, this.currentWidth, this.currentSize, 0, 0,
Math.PI * 2);
ctx.fill();

this.currentWidth * 0.6);

cg.addColorStop(1, `hsla(${hue},${sat}%,${light}%,0)`);

ctx.beginPath();
ctx.ellipse(0, 0, this.currentWidth * 0.6, this.currentSize *
0.4, 0, 0, Math.PI * 2);
ctx.fill();
}

}

}
for (let i = 0; i < MAX_PARTICLES; i++) particles.push(new Particle());

}

const logs = [
{ x: 0, y: 30, len: 260, r: 18, angle: 0.05 },
{ x: -30, y: 10, len: 220, r: 16, angle: -0.35 },
{ x: 35, y: 8, len: 230, r: 15, angle: 0.3 },
{ x: 5, y: -5, len: 180, r: 13, angle: -0.12 },
];
// Ð“ÐµÐ½ÐµÑ€Ð¸Ñ€ÑƒÐµÐ¼ Â«Ñ‚Ñ€ÐµÑ‰Ð¸Ð½Ñ‹Â» Ð´Ð»Ñ• ÐºÐ°Ð¶Ð´Ð¾Ð³Ð¾
Ð±Ñ€ÐµÐ²Ð½Ð° Ð¾Ð´Ð¸Ð½ Ñ€Ð°Ð·

// Ð¢Ð»ÐµÑŽÑ‰Ð¸Ðµ Ð¿ÑÑ‚Ð½Ð°

}

// Ð¢ÐµÐ»Ð¾ Ð±Ñ€ÐµÐ²Ð½Ð°

function drawFireplace(time) {
ctx.fillStyle = '#0a0503';
ctx.fillRect(0, 0, W, H);

shade)},${Math.floor(32 * shade)})`;

}

ctx.fillRect(wallLeft, mantelY, wallRight - wallLeft, H - mantelY);
}

function loop(time) {

}
